    Sub P19_TimeoutZero()
        LoadFresh("p19", Mutated(Sub(o) o("network")("request_timeout_seconds") = 0))
        For attempt = 1 To 2
            Try
                DeribitClient.GetCandlesAsync("1", 5).GetAwaiter().GetResult()
                Console.WriteLine("  attempt " & attempt & ": returned")
            Catch ex As Exception
                Console.WriteLine("  attempt " & attempt & " (timeout key now " & SettingsLoader.Current.Network.RequestTimeoutSeconds & "): " &
                                  ex.GetType().Name & " <- " & If(ex.InnerException?.GetType().Name, "") & ": " & If(ex.InnerException?.Message, ex.Message).Split(ChrW(10))(0))
            End Try
            LoadFresh("p19b", File.ReadAllText(RepoSettings))   ' fix the file: 15 s
        Next
    End Sub

    Sub Run(name As String, body As Action)
        Console.WriteLine()
        Console.WriteLine("=== " & name & " ===")
        Try
            body()
        Catch ex As Exception
            Console.WriteLine("  UNCAUGHT " & ex.GetType().Name & ": " & ex.Message)
        End Try
    End Sub

    ' ---------------------------------------------------------------- helpers
    Function Mutated(action As Action(Of JsonObject)) As String
        Dim root = JsonNode.Parse(File.ReadAllText(RepoSettings)).AsObject()
        action(root)
        Return root.ToJsonString(New JsonSerializerOptions With {.WriteIndented = True})
    End Function

    Function Dir(name As String) As String
        Dim d = Path.Combine(workRoot, name)
        Directory.CreateDirectory(d)
        Return d
    End Function

    ' Simulates a fresh process: the loader's _current is a New EngineSettings() at startup.
    Sub ResetLoaderToProcessStart()
        Dim t = GetType(SettingsLoader)
        t.GetField("_current", BindingFlags.NonPublic Or BindingFlags.Static).SetValue(Nothing, New EngineSettings())
        t.GetField("_lastLoadError", BindingFlags.NonPublic Or BindingFlags.Static).SetValue(Nothing, "")
    End Sub

    Function LoadFresh(dirName As String, baseJson As String, Optional overlayJson As String = Nothing) As EngineSettings
        Dim d = Dir(dirName)
        File.WriteAllText(Path.Combine(d, "settings.json"), baseJson)
        Dim ov = Path.Combine(d, "settings.local.json")
        If overlayJson IsNot Nothing Then File.WriteAllText(ov, overlayJson) Else If File.Exists(ov) Then File.Delete(ov)
        ResetLoaderToProcessStart()
        SettingsLoader.Initialise(Path.Combine(d, "settings.json"))
        Return SettingsLoader.Current
    End Function

    Function Candles(n As Integer, startPx As Double, stepFn As Func(Of Integer, Double),
                     rangeFn As Func(Of Integer, Double), Optional barSec As Integer = 60,
                     Optional volFn As Func(Of Integer, Double) = Nothing) As List(Of Candle)
        Dim res As New List(Of Candle)
        Dim px As Double = startPx
        Dim t0 As Long = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() - CLng(n) * barSec * 1000
        For i = 0 To n - 1
            Dim o As Double = px
            Dim c As Double = px + stepFn(i)
            Dim rg As Double = rangeFn(i)
            res.Add(New Candle With {.Timestamp = t0 + CLng(i) * barSec * 1000, .Open = o, .Close = c,
                                     .High = Math.Max(o, c) + rg / 2, .Low = Math.Min(o, c) - rg / 2,
                                     .Volume = If(volFn Is Nothing, 10 + (i Mod 7) * 3, volFn(i))})
            px = c
        Next
        Return res
    End Function

    Sub P17_FlushTrace()
        ' NY 1-min: 240 quiet bars (TR ~40) then a 2-bar flush of ~400 each.
        Dim cs = Candles(250, 62700, Function(i) If(i >= 248, -350.0, If(i Mod 2 = 0, 6.0, -6.0)),
                         Function(i) If(i >= 248, 60.0, 34.0))
        Dim atr = IndicatorEngine.CalcATR(cs, 7)
        Dim px = cs.Last().Close
        Dim cfg As New EngineSettings()
        Dim r As New IndicatorResults With {.CurrentPrice = px, .ATR = atr, .SessionUtcHour = 14,
                                            .SwingStopShort = px + 900, .SwingTargetShort = 0,
                                            .SwingStopLong = 0, .SwingTargetLong = px + 900}
        Dim s = Lv(r, cfg, False) : Dim l = Lv(r, cfg, True)
        Console.WriteLine(String.Format(Inv, "  price {0:F1}  ATR(7,1m) after flush={1:F1} ({2})  floor={3:F1} USD",
            px, atr, Bps(atr, px), cfg.Scoring.TradeCosts.EffectiveMinMovePct * px))
        Console.WriteLine(String.Format(Inv, "  SHORT: stop {0} +{1:F1} ({2}) target {3} -{4:F1} ({5}) | LONG: stop {6} -{7:F1} target {8} +{9:F1}",
            s.StopReason, s.StopPx - px, Bps(s.StopPx - px, px), s.TargetReason, px - s.Target, Bps(px - s.Target, px),
            l.StopReason, px - l.StopPx, l.TargetReason, l.Target - px))
        Dim v As New VerdictResult With {.Verdict = "STRONG SHORT", .Confidence = "HIGH"}
        ScoringEngine.CalcKellySizing(v, atr * cfg.Scoring.AtrStopMultiplier, px, cfg)
        Dim realRisk = v.KellyContracts * 10.0 * (s.StopPx - px) / px
        Dim fees = v.KellyContracts * 10.0 * 5 / 10000.0
        Console.WriteLine(String.Format(Inv, "  Kelly HIGH: contracts={0} risk_usd={1:F2} lev_capped={2}; actual stop-out = {3:F2} + fees {4:F2} = {5:F2} USD",
            v.KellyContracts, v.KellyRiskUsd, v.KellyLevCapped, realRisk, fees, realRisk + fees))
        Dim grossB = (px - s.Target) / (s.StopPx - px)
        Dim netB = ((px - s.Target) - px * 3 / 10000.0) / ((s.StopPx - px) + px * 5 / 10000.0)
        Console.WriteLine(String.Format(Inv, "  b: Kelly uses {0:F3}; placed gross {1:F3}; placed net (maker/maker win, maker/taker loss) {2:F3}; breakeven p: {3:P1} vs {4:P1}",
            cfg.Scoring.AtrTargetMultiplier / cfg.Scoring.AtrStopMultiplier, grossB, netB, 1 / (1 + cfg.Scoring.AtrTargetMultiplier / cfg.Scoring.AtrStopMultiplier), 1 / (1 + netB)))
        For Each p In {0.55, 0.65}
            Console.WriteLine(String.Format(Inv, "    p={0}: f* Kelly-shown={1:F3}  f* net={2:F3}", p,
                ((cfg.Scoring.AtrTargetMultiplier / cfg.Scoring.AtrStopMultiplier) * p - (1 - p)) / (cfg.Scoring.AtrTargetMultiplier / cfg.Scoring.AtrStopMultiplier), (netB * p - (1 - p)) / netB))
        Next
        ' Same thing on a quiet NY tape (ATR 40)
        Dim q As New IndicatorResults With {.CurrentPrice = 62000, .ATR = 40, .SessionUtcHour = 14}
        Dim qs = Lv(q, cfg, False)
        Dim qNet = ((62000 - qs.Target) - 62000 * 3 / 10000.0) / ((qs.StopPx - 62000) + 62000 * 5 / 10000.0)
        Console.WriteLine(String.Format(Inv, "  quiet NY (ATR 40): placed net b={0:F3}; p=0.55 f* net={1:F3}; p=0.65 f* net={2:F3} (Kelly shows 0.139 / 0.330, both capped to 5 %)",
            qNet, (qNet * 0.55 - 0.45) / qNet, (qNet * 0.65 - 0.35) / qNet))
    End Sub
