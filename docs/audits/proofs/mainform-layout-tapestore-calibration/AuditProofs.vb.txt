Imports System.Globalization

Module AuditProofs

    ' P1 — LivePerformanceTracker.UpdateAsync awaits _initTcs BEFORE its Enabled check.
    ' If InitialiseAsync is never called (MainForm ctor gates it on performance_display.enabled),
    ' UpdateAsync never completes. Expect: "HANGS".
    Sub P1_UpdateAsyncHangsWithoutInit()
        Dim t As Task = LivePerformanceTracker.UpdateAsync(New VerdictResult(), New IndicatorResults(),
                                                           Nothing, DateTime.UtcNow)
        Dim done As Boolean = t.Wait(TimeSpan.FromSeconds(3))
        Console.WriteLine("P1: " & If(done, "completed", "HANGS (RunAnalysisAsync never reaches EmitBridgeSignal)"))
    End Sub

    ' P2 — CommitMinNetMove validation admits NaN; the Step-5c gate then fails open.
    Sub P2_NaNPassesValidationAndDisablesGate()
        Dim entered As Double
        Dim parsed = Double.TryParse("NaN", NumberStyles.Float, CultureInfo.InvariantCulture, entered)
        Dim rejected As Boolean = (Not parsed) OrElse entered < 0.0 OrElse entered > 0.01
        Console.WriteLine("P2a: parsed=" & parsed & " rejected=" & rejected)          ' expect parsed=True rejected=False
        Dim floorDist As Double = (0.0003 + entered) * 60000.0
        Dim effDist As Double = 5.0                                                    ' a $5 target — far under any sane floor
        Dim gateFires As Boolean = floorDist > 0 AndAlso effDist < floorDist           ' ScoringEngine_Calculate_Verdict.vb:316
        Console.WriteLine("P2b: gateFires=" & gateFires)                               ' expect False
        Try
            System.Text.Json.JsonSerializer.Serialize(New With {.v = entered})
            Console.WriteLine("P2c: serialized")
        Catch ex As ArgumentException
            Console.WriteLine("P2c: Save throws -> singleton keeps NaN, file keeps old value")   ' expect this
        End Try
    End Sub

    ' P3 — BuildWsStatusSegment: CInt over (UtcNow - DateTime.MinValue).TotalSeconds.
    ' Reached when IsConnected=True before the first frame stamps _lastFrameUtc.
    Sub P3_WsSegmentOverflow()
        Try
            Dim ageSec As Integer = CInt(Math.Max(0, (DateTime.UtcNow - DateTime.MinValue).TotalSeconds))
            Console.WriteLine("P3: no throw, ageSec=" & ageSec)
        Catch ex As OverflowException
            Console.WriteLine("P3: OverflowException -> escapes UpdateLogInfo -> RunAnalysisAsync -> modal MessageBox")
        End Try
    End Sub

    ' P4 — CalcMTFGate fails OPEN on missing 15m data.
    Sub P4_MtfGateFailsOpen()
        Dim tr As String = "", al As String = "", det As String = ""
        Dim adx As Double, pl As Boolean, ps As Boolean
        IndicatorEngine.CalcMTFGate(Nothing, tr, adx, al, pl, ps, det,
                                    adxPeriod:=14, adxMin:=20, minOf:=2, candleLookback:=60)   ' MECHANISM literals: any values reach the Nothing guard
        Console.WriteLine("P4: passLong=" & pl & " passShort=" & ps & " (" & det & ")")      ' expect True/True
    End Sub

End Module
