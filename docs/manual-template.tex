%% Override Pandoc's Shaded environment (fenced code blocks).
%% Defensive: Pandoc only pre-defines Shaded when its default highlighting
%% is active and code blocks are present. Define if absent, renew if present.
\makeatletter
\@ifundefined{Shaded}{%
  \newenvironment{Shaded}{%
    \begin{tcolorbox}[codeblock]%
    \small\ttfamily\color{bodytext}%
  }{%
    \end{tcolorbox}%
  }
}{%
  \renewenvironment{Shaded}{%
    \begin{tcolorbox}[codeblock]%
    \small\ttfamily\color{bodytext}%
  }{%
    \end{tcolorbox}%
  }
}
\makeatother
